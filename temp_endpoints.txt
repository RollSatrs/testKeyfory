
// Завершить заказ
executerRoute.put('/complete-order/:executionId', async (req, res) => {
  try {
    const { executionId } = req.params;

    console.log('\n === ЗАВЕРШЕНИЕ ЗАКАЗА ===');
    console.log(' ID выполнения:', executionId);

    const ServiceExecution = require('../../../database/dbTables').ServiceExecution;

    const execution = await ServiceExecution.findByPk(executionId);
    if (!execution) {
      return res.status(404).json({ error: 'Выполнение услуги не найдено' });
    }

    execution.status = 'completed';
    execution.completed_at = new Date();
    await execution.save();

    console.log(' Заказ завершен:', executionId);
    res.json(execution);
  } catch (error) {
    console.error(' Ошибка завершения заказа:', error);
    res.status(500).json({ error: error.message || 'Ошибка сервера' });
  }
});

// Отменить заказ
executerRoute.put('/cancel-order/:executionId', async (req, res) => {
  try {
    const { executionId } = req.params;
    const { reason } = req.body;

    console.log('\n === ОТМЕНА ЗАКАЗА ===');    
    console.log(' ID выполнения:', executionId);
    console.log(' Причина:', reason);

    const ServiceExecution = require('../../../database/dbTables').ServiceExecution;

    const execution = await ServiceExecution.findByPk(executionId);
    if (!execution) {
      return res.status(404).json({ error: 'Выполнение услуги не найдено' });
    }

    execution.status = 'cancelled';
    execution.cancellation_reason = reason;
    execution.cancelled_at = new Date();
    await execution.save();

    console.log(' Заказ отменен:', executionId);
    res.json(execution);
  } catch (error) {
    console.error(' Ошибка отмены заказа:', error);
    res.status(500).json({ error: error.message || 'Ошибка сервера' });
  }
});

// Запросить замену материала
executerRoute.post('/request-replacement', async (req, res) => {
  try {
    const { execution_id, material_id, reason } = req.body;

    console.log('\n === ЗАПРОС ЗАМЕНЫ МАТЕРИАЛА ===');
    console.log(' Данные:', { execution_id, material_id, reason });

    const request = {
      id: Date.now(),
      execution_id,
      material_id,
      reason,
      status: 'pending',
      created_at: new Date()
    };

    console.log(' Запрос на замену создан:', request.id);
    res.json(request);
  } catch (error) {
    console.error(' Ошибка создания запроса на замену:', error);
    res.status(500).json({ error: error.message || 'Ошибка сервера' });
  }
});
